## 集成方式总览

### 集成方式对比

| 方式 | 适用场景 | 优点 | 缺点 |
|------|----------|------|------|
| SDK集成 | 有官方SDK的服务 | 功能完整、稳定性好 | 包体积增加、版本依赖 |
| REST API | 无SDK或轻量集成 | 灵活、无依赖 | 需自己处理签名、错误 |
| Webhook | 异步回调场景 | 实时性高、解耦 | 需公网接收端 |
| SDK + API混合 | 复杂业务场景 | 兼顾灵活性和完整性 | 结构复杂 |

### 集成决策树

```
是否需要集成第三方服务
    │
    ├─► 是否有官方SDK？
    │       │
    │       ├─► SDK是否维护活跃？──► 否 ──► 使用REST API
    │       │
    │       └─► 是 ──► 是否需要全部功能？
    │               │
    │               ├─► 是 ──► 使用完整SDK
    │               │
    │               └─► 否 ──► 使用SDK核心模块 + API补充
    │
    └─► 是否需要异步回调？
            │
            ├─► 是 ──► 配置WebHook接收服务
            │
            └─► 否 ──► 同步调用即可
```

```javascript
const client = new OSS({

  accessKeySecret: process.env.OSS_ACCESS_KEY_SECRET,

  region: process.env.OSS_REGION
});

// 生成签名URL（有效期10分钟）
async function generateUploadToken(fileName, expires = 600) {
  const result = await client.signatureUrl(fileName, {
    expires,
    method: 'PUT'
  });
  return {
    uploadUrl: result,
    fileName,
    expires
  };
}

// 生成文件夹内的文件路径
function generateFilePath(dir, fileName, userId) {
  const timestamp = Date.now();
  const ext = path.extname(fileName);
  return `${dir}/${userId}/${timestamp}${ext}`;
}

async function uploadToOSS(uploadUrl, file) {
  return fetch(uploadUrl, {
    method: 'PUT',
    body: file,
    headers: {
      'Content-Type': file.type
    }
  });
}
```

### 后端直接上传

适用于后端代理上传场景

```javascript
const OSS = require('ali-oss');
const path = require('path');

class AliyunOSSService {
  constructor() {
    this.client = new OSS({
      accessKeyId: process.env.OSS_ACCESS_KEY_ID,
      accessKeySecret: process.env.OSS_ACCESS_KEY_SECRET,
      bucket: process.env.OSS_BUCKET,
      region: process.env.OSS_REGION,
      timeout: 30000
    });
  }

  // 上传文件
  async upload(filePath, localFilePath) {
    try {
      const result = await this.client.put(filePath, localFilePath);
      logger.info('OSS上传成功', {
        filePath,
        etag: result.etag,
        url: result.url
      });
      return {
        success: true,
        url: result.url,
        etag: result.etag
      };
    } catch (error) {
      logger.error('OSS上传失败', { filePath, error: error.message });
      throw error;
    }
  }

  // 下载文件
  async download(filePath, localSavePath) {
    try {
      const result = await this.client.get(filePath, localSavePath);
      return { success: true, path: localSavePath };
    } catch (error) {
      logger.error('OSS下载失败', { filePath, error: error.message });
      throw error;
    }
  }

  // 删除文件
  async delete(filePath) {
    try {
      await this.client.delete(filePath);
      logger.info('OSS文件删除成功', { filePath });
      return { success: true };
    } catch (error) {
      logger.error('OSS文件删除失败', { filePath, error: error.message });
      throw error;
    }
  }

  // 批量删除
  async deleteMulti(filePaths) {
    try {
      const result = await this.client.deleteMulti(filePaths);
      return { success: true, deleted: result.deleted };
    } catch (error) {
      logger.error('OSS批量删除失败', { filePaths, error: error.message });
      throw error;
    }
  }

  // 判断文件是否存在
  async exists(filePath) {
    try {
      await this.client.get(filePath);
      return true;
    } catch (error) {
      if (error.code === 'NoSuchKey') return false;
      throw error;
    }
  }

  // 获取文件元信息
  async getMeta(filePath) {
    try {
      const result = await this.client.getMeta(filePath);
      return {
        size: result.size,
        lastModified: result.lastModified,
        etag: result.etag
      };
    } catch (error) {
      logger.error('获取OSS文件元信息失败', { filePath, error: error.message });
      throw error;
    }
  }

  // 生成访问URL
  getPublicUrl(filePath) {
    return `https://${process.env.OSS_BUCKET}.${process.env.OSS_REGION}.aliyuncs.com/${filePath}`;
  }

  // 生成签名URL（私有读取）
  async getSignUrl(filePath, expires = 3600) {
    return await this.client.signatureUrl(filePath, { expires });
  }
}

module.exports = new AliyunOSSService();
```

### 错误码处理

| 错误码 | 含义 | 处理方式 |
|--------|------|----------|
| NoSuchBucket | bucket不存在 | 检查配置 |
| NoSuchKey | 文件不存在 | 返回404或上传新文件 |
| AccessDenied | 权限不足 | 检查RAM权限策略 |
| RequestTimeTooSkewed | 请求时间偏移过大 | 同步服务器时间 |
| TooManyObjects | bucket内对象过多 | 清理或归档历史文件 |

### 支付流程

```
┌─────────────────────────────────────────────────────────────┐
│                        JSAPI支付流程                          │
├─────────────────────────────────────────────────────────────┤
│  1. 用户下单 → 后端创建订单                                    │
│  2. 后端调用微信预下单API /v3/pay/transactions/jsapi         │
│  3. 返回 prepay_id                                           │
│  4. 前端调用 wx.requestPayment() 调起支付                    │
│  5. 用户完成支付                                              │
│  6. 微信异步回调 → 后端接收通知                                │
│  7. 验证签名 → 更新订单状态                                    │
│  8. 返回 SUCCESS 给微信                                      │
└─────────────────────────────────────────────────────────────┘
```

### SDK安装与使用


```javascript
const WeChatPay = require('wechatpayv3');
const axios = require('axios');

class WeChatPayService {
  constructor() {
    this.client = new WeChatPay({
      mchId: process.env.WX_MCH_ID,
      serialNo: process.env.WX_SERIAL_NO,
      privateKey: process.env.WX_PRIVATE_KEY,
      apiv3Key: process.env.WX_APIV3_KEY,
      certPath: process.env.WX_CERT_PATH
    });
  }

  // ========== JSAPI支付：统一下单 ==========
  async createJsapiOrder(params) {
    const { orderId, amount, openid, description } = params;
    
    try {
      const result = await this.client.pay({
        description,
        out_trade_no: orderId,
        time_expire: this.getExpireTime(30), // 30分钟后过期
        notify_url: process.env.WX_NOTIFY_URL,
        amount: {
          total: Math.round(amount * 100), // 金额：分
          currency: 'CNY'
        },
        payer: {
          openid
        },
        attach: JSON.stringify({ orderId }) // 附加数据，微信会原样返回
      });

      logger.info('微信JSAPI下单成功', {
        orderId,
        prepayId: result.prepay_id
      });

      return {
        success: true,
        prepayId: result.prepay_id
      };
    } catch (error) {
      logger.error('微信JSAPI下单失败', {
        orderId,
        error: error.message,
        code: error.code
      });
      throw error;
    }
  }

  // ========== 前端调起支付参数 ==========
  getJsapiPayParams(appId, prepayId) {
    const timestamp = Math.floor(Date.now() / 1000).toString();
    const nonceStr = this.generateNonceStr();
    const packageStr = `prepay_id=${prepayId}`;

    // 签名（前端使用）
    const paySign = this.sign([
      appId,
      timestamp,
      nonceStr,
      packageStr
    ].join('\n'), process.env.WX_APIV3_KEY);

    return {
      appId,
      timestamp,
      nonceStr,
      package: packageStr,
      paySign,
      signType: 'RSA'
    };
  }

  // ========== 查询订单 ==========
  async queryOrder(tradeNo) {
    try {
      const result = await this.client.query({
        out_trade_no: tradeNo
      });
      return result;
    } catch (error) {
      logger.error('微信订单查询失败', { tradeNo, error: error.message });
      throw error;
    }
  }

  // ========== 关闭订单 ==========
  async closeOrder(tradeNo) {
    try {
      await this.client.close({
        out_trade_no: tradeNo
      });
      return { success: true };
    } catch (error) {
      logger.error('微信关闭订单失败', { tradeNo, error: error.message });
      throw error;
    }
  }

  // ========== 申请退款 ==========
  async refund(params) {
    const { tradeNo, refundAmount, reason } = params;
    
    try {
      const result = await this.client.refund({
        out_trade_no: tradeNo,
        out_refund_no: `REF${tradeNo}${Date.now()}`,
        amount: {
          refund: Math.round(refundAmount * 100),
          total: Math.round(refundAmount * 100), // 需从订单获取实际值
          currency: 'CNY'
        },
        reason
      });
      
      logger.info('微信退款申请成功', {
        tradeNo,
        refundId: result.refund_id
      });

      return {
        success: true,
        refundId: result.refund_id
      };
    } catch (error) {
      logger.error('微信退款申请失败', { tradeNo, error: error.message });
      throw error;
    }
  }

  // ========== 回调通知处理 ==========
  async verifyCallback(notify) {
    try {
      // 验证签名
      const verified = await this.client.verify(notify.signature, notify);
      if (!verified) {
        throw new Error('回调签名验证失败');
      }

      // 解密通知数据
      const decrypted = this.client.decryptNotify(notify);
      
      // 处理通知
      const { transaction_id, trade_state, out_trade_no, amount } = decrypted;
      
      logger.info('微信支付回调', {
        tradeNo: out_trade_no,
        state: trade_state,
        transactionId: transaction_id
      });

      return {
        success: true,
        data: {
          tradeNo: out_trade_no,
          transactionId: transaction_id,
          tradeState: trade_state,
          amount: amount.payer_total / 100
        }
      };
    } catch (error) {
      logger.error('微信回调处理失败', { error: error.message });
      throw error;
    }
  }

  // ========== 工具方法 ==========
  getExpireTime(minutes) {
    const expire = new Date(Date.now() + minutes * 60 * 1000);
    return expire.toISOString().replace(/\.\d{3}Z$/, '+0800');
  }

  generateNonceStr(length = 32) {
    const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
    let result = '';
    for (let i = 0; i < length; i++) {
      result += chars.charAt(Math.floor(Math.random() * chars.length));
    }
    return result;
  }

  sign(message, key) {
    const crypto = require('crypto');
    return crypto.createHmac('sha256', key)
      .update(message)
      .digest('hex');
  }
}

module.exports = new WeChatPayService();
```

### 支付结果处理

```javascript
// router/post('/payment/callback')
async function handlePaymentCallback(ctx) {
  const notify = ctx.request.body;
  
  try {
    const result = await wechatPayService.verifyCallback(notify);
    
    if (result.success) {
      const { tradeNo, transactionId, tradeState } = result.data;
      
      // 根据交易状态处理
      switch (tradeState) {
        case 'SUCCESS':
          await orderService.updatePayStatus(tradeNo, 'PAID', transactionId);
          break;
        case 'REFUND':
          await orderService.updatePayStatus(tradeNo, 'REFUNDED', transactionId);
          break;
        case 'PAYERROR':
          await orderService.updatePayStatus(tradeNo, 'FAILED');
          break;
        default:
          logger.warn('未知支付状态', { tradeState });
      }
      
      // 返回SUCCESS给微信
      ctx.body = { code: 'SUCCESS', message: '成功' };
    }
  } catch (error) {
    logger.error('支付回调处理异常', { error: error.message });
    ctx.body = { code: 'FAIL', message: '失败' };
  }
}
```

### 注意事项

- **证书安全存储**: 私钥和证书必须存放在安全位置，禁止硬编码
- **回调地址**: 必须公网可访问，建议使用HTTPS
- **签名验证**: 所有回调必须验证签名，防止伪造
- **幂等处理**: 微信可能发送多次回调，需保证幂等性
- **敏感数据**: 使用AEAD_AES_256_GCM加密传输敏感信息

### 错误码处理

| 错误码 | 含义 | 处理方式 |
|--------|------|----------|
| ORDERPAID | 订单已支付 | 查询订单状态，避免重复操作 |
| ORDERCLOSED | 订单已关闭 | 提示用户重新下单 |
| SYSTEMERROR | 系统错误 | 重试或人工介入 |
| BANKERROR | 银行异常 | 提示用户换支付方式 |
| USERPAYING | 用户支付中 | 轮询查询订单状态 |

```javascript
const AlipaySdk = require('alipay-sdk').default;

class AlipayService {
  constructor() {
    this.client = new AlipaySdk({
      appId: process.env.ALIPAY_APP_ID,
      privateKey: process.env.ALIPAY_PRIVATE_KEY,
      alipayPublicKey: process.env.ALIPAY_PUBLIC_KEY,
      gateway: process.env.NODE_ENV === 'production'
        ? 'https://openapi.alipay.com/gateway.do'
        : 'https://openapi-sandbox.dl.alipaydev.com/gateway.do'
    });

  // 当面付：条码支付
  async payWithBarcode(params) {
    const { tradeNo, authCode, amount, subject } = params;
    
    try {
      const result = await this.client.exec('alipay.trade.pay', {
        out_trade_no: tradeNo,
        scene: 'bar_code',
        auth_code: authCode,
        subject,
        total_amount: amount,
        currency: 'CNY',
        notify_url: process.env.ALIPAY_NOTIFY_URL
      });

      logger.info('支付宝条码支付成功', { tradeNo, result });

      return {
        success: true,
        tradeNo: result.trade_no
      };
    } catch (error) {
      logger.error('支付宝条码支付失败', { tradeNo, error: error.message });
      throw error;
    }
  }

  // 电脑网站支付
  async createWebPayOrder(params) {

        currency: 'CNY',

      out_trade_no: tradeNo
    });

  }
```

```javascript
    // 频率限制：检查缓存
    const cacheKey = `sms:verify:${phone}`;
    const lastSend = await redis.get(cacheKey);
    
    if (lastSend) {
      throw new BusinessError('短信发送过于频繁，请稍后再试');
    }

    const result = await this.sendSMS({
      phoneNumbers: phone,
      templateParam: { code }
    });

    if (result.success) {
      // 设置发送间隔缓存（60秒）
      await redis.set(cacheKey, '1', 'EX', 60);
      // 设置验证码有效期缓存（5分钟）

    const cachedCode = await redis.get(`sms:code:${phone}`);

      return { valid: false, reason: '验证码已过期' };
    }

    if (cachedCode !== code) {
      return { valid: false, reason: '验证码错误' };
    }

    // 验证成功后删除
    await redis.del(`sms:code:${phone}`);
    
    return { valid: true };
```

```javascript
class CircuitBreakerManager {
  constructor() {
    this.breakers = new Map();
  }

  // 创建熔断器
  createBreaker(name, options = {}) {
    const breaker = new CircuitBreaker(async (fn, ...args) => fn(...args), {
      name,
      timeout: options.timeout || 10000,      // 操作超时时间
      errorThresholdPercentage: 50,           // 错误率阈值
      resetTimeout: 30000,                    // 重置时间
      halfOpenTime: 10000,                    // 半开状态尝试时间
      ...options

    });

    breaker.on('close', () => {
      logger.info('circuit_breaker_close', { name });
    });

    breaker.on('halfOpen', () => {
      logger.info('circuit_breaker_half_open', { name });
    });

    this.breakers.set(name, breaker);
    return breaker;
  }

  // 获取熔断器
  getBreaker(name) {
    return this.breakers.get(name);
  }

  // 通用包装方法
  async execute(name, fn, fallback = null, ...args) {
    let breaker = this.breakers.get(name);

    if (!breaker) {
      breaker = this.createBreaker(name);
    }

    if (fallback) {
      return breaker.fire(fn, ...args).catch(async (error) => {
        logger.warn('circuit_breaker_fallback', { name, error: error.message });
        return await fallback(...args);
      });
    }

    return breaker.fire(fn, ...args);
  }
}
```

```javascript
// ========== 重试装饰器 ==========
function withRetry(fn, options = {}) {
  return async function(...args) {
    return retry(async (bail) => {
      try {
        return await fn(...args);
      } catch (error) {
        // 不重试的错误类型
        if (error.noRetry) {
          bail(error);
          return;
        }
        throw error;
      }
    }, {
      retries: options.retries || 3,
      minTimeout: options.minTimeout || 1000,
      maxTimeout: options.maxTimeout || 10000,
      factor: options.factor || 2,
      onRetry: (error) => {
        logger.warn('retry_attempt', {
          fn: fn.name,
          error: error.message,
          attempt: error.attemptNumber
        });
      },
      ...options
    });
  };
}

// ========== 超时包装 ==========
const withTimeout = (promise, ms) => {
  return new Promise((resolve, reject) => {
    const timer = setTimeout(() => {
      reject(new ThirdPartyError('timeout', 'REQUEST_TIMEOUT', `请求超时：${ms}ms`));
    }, ms);

    promise
      .then(resolve)
      .catch(reject)
      .finally(() => clearTimeout(timer));
  });
};

// ========== 组合使用 ==========
class ReliableService {
  constructor(service) {
    this.service = service;
  }

  async upload(...args) {
    const fn = withRetry(
      (filePath, localFilePath) => withTimeout(
        this.service.upload(filePath, localFilePath),
        30000
      ),
      { retries: 3 }
    );
    return fn(...args);
  }
}
```

```javascript
async function checkThirdPartyHealth() {
  const checks = [
    { name: 'aliyun-oss', fn: checkOSS },
    { name: 'wechat-pay', fn: checkWechatPay },
    { name: 'aliyun-sms', fn: checkAliSMS },
    { name: 'jiguang-push', fn: checkJiguang }
  ];

  const results = await Promise.allSettled(
    checks.map(({ name, fn }) => fn().then(r => ({ name, ...r })))
  );

  return results.map((result, index) => {
    const { name, fn } = checks[index];
    if (result.status === 'fulfilled') {
      return { name, healthy: result.value.healthy, latency: result.value.latency };
    }
    return { name, healthy: false, error: result.reason.message };
  });
}

  }
}
```

## 常见问题

### Q: 第三方服务响应慢怎么办？

A: 设置合理的超时时间，使用熔断器防止级联失败，必要时返回降级数据。

### Q: 如何保证消息不丢失？

A: 1) 支付回调等关键操作使用事务；2) 记录完整的调用日志；3) 定期对账。

### Q: 敏感信息如何管理？

A: 1) 使用环境变量或密钥管理服务；2) 禁止硬编码；3) 定期轮换密钥。

### Q: 如何处理跨地域访问延迟？

A: 1) 选择最近的接入点；2) 使用CDN加速；3) 异步处理非实时需求。

### Q: 如何监控第三方服务可用性？

A: 1) 记录调用成功率和响应时间；2) 配置熔断器状态监控；3) 设置告警阈值。
